class Solution {
    public int minDominoRotations(int[] tops, int[] bottoms) {
        int result=check(tops[0],tops,bottoms);
        if(result!=-1){
            return result;
        }
        if(tops[0]!=bottoms[0]){
            int result1=check(bottoms[0],tops,bottoms);
            if(result1!=-1){
                return result1;
            }
        }
        return -1;
    }
    public int check(int candidate,int[] tops,int [] bottoms){
        int topcount=0;
        int bottomcount=0;
        for(int i=0;i<tops.length;i++){
            if(tops[i]!=candidate && bottoms[i]!=candidate){
                return -1;
            }
            if(tops[i]!=candidate){
                topcount++;
            }
            if(bottoms[i]!=candidate){
                bottomcount++;
            }
        }
        return Math.min(topcount,bottomcount);
    }
}
